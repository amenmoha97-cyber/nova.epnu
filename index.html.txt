        function login() {
            const name = document.getElementById('studentName').value.trim();
            const national = document.getElementById('studentNational').value.trim();
            const id = document.getElementById('studentId').value.trim();
            if (!name || !national || !id) {
                alert('من فضلك املأ جميع البيانات');
                return;
            }
            document.getElementById('displayName').innerText = name;
            document.getElementById('loginPage').style.display = 'none';
            document.getElementById('dashboard').style.display = 'flex';
            initDashboard();
        }

        function logout() {
            document.getElementById('loginPage').style.display = 'flex';
            document.getElementById('dashboard').style.display = 'none';
        }

        function switchTab(tab) {
            document.querySelectorAll('.page-section').forEach(el => el.classList.remove('active'));
            document.getElementById('section-' + tab).classList.add('active');
            document.querySelectorAll('.sidebar .tab-btn').forEach(el => el.classList.remove('active'));
            event.target.closest('.tab-btn').classList.add('active');
        }

        function switchAddTab(type) {
            if (type === 'add') {
                document.getElementById('addGroup').style.display = 'block';
                document.getElementById('noAddGroup').style.display = 'none';
            } else {
                document.getElementById('addGroup').style.display = 'none';
                document.getElementById('noAddGroup').style.display = 'block';
            }
            document.querySelectorAll('.sub-tabs button').forEach(b => b.classList.remove('active'));
            event.target.classList.add('active');
        }

        function initDashboard() {
            renderSubjects('coreSubjectsGrid', coreSubjects, 'core');
            renderSubjects('addSubjectsGrid', addSubjects, 'add');
            renderSubjects('noAddSubjectsGrid', noAddSubjects, 'noAdd');
            initPortfolio();
        }

        function renderSubjects(containerId, subjects, type) {
            const container = document.getElementById(containerId);
            container.innerHTML = '';
            subjects.forEach((sub, index) => {
                const card = document.createElement('div');
                card.className = 'subject-card';
                card.innerHTML = `
                    <span class="icon">${sub.icon}</span>
                    <div class="name">${sub.name}</div>
                `;
                card.onclick = () => openSubject(sub.name, type, index);
                container.appendChild(card);
            });
        }

        function openSubject(name, type, index) {
            let bContainer, bTabs, bContent;
            if (type === 'core') {
                currentCoreSubject = name;
                bContainer = document.getElementById('bContainerCore');
                bTabs = document.getElementById('bTabsCore');
                bContent = document.getElementById('bContentCore');
                bContainer.style.display = 'block';
                document.getElementById('bContainerAdd').style.display = 'none';
                document.getElementById('bContainerNoAdd').style.display = 'none';
            } else if (type === 'add') {
                currentAddSubject = name;
                bContainer = document.getElementById('bContainerAdd');
                bTabs = document.getElementById('bTabsAdd');
                bContent = document.getElementById('bContentAdd');
                bContainer.style.display = 'block';
                document.getElementById('bContainerCore').style.display = 'none';
                document.getElementById('bContainerNoAdd').style.display = 'none';
            } else if (type === 'noAdd') {
                currentNoAddSubject = name;
                bContainer = document.getElementById('bContainerNoAdd');
                bTabs = document.getElementById('bTabsNoAdd');
                bContent = document.getElementById('bContentNoAdd');
                bContainer.style.display = 'block';
                document.getElementById('bContainerCore').style.display = 'none';
                document.getElementById('bContainerAdd').style.display = 'none';
            }

            bTabs.innerHTML = '';
            for (let i = 1; i <= 6; i++) {
                const btn = document.createElement('button');
                btn.innerText = `B${i}`;
                btn.dataset.b = `B${i}`;
                if (i === 1) btn.classList.add('active');
                btn.onclick = function() {
                    bTabs.querySelectorAll('button').forEach(b => b.classList.remove('active'));
                    this.classList.add('active');
                    loadBFiles(bContent, `B${i}`);
                };
                bTabs.appendChild(btn);
            }
            loadBFiles(bContent, 'B1');
        }

        function loadBFiles(container, bLabel) {
            let html = `<h4 style="color:#0d0d0d; margin-bottom:12px;">📂 ${bLabel} - الملفات</h4>`;
            dummyFiles.forEach((file, idx) => {
                let icon = 'fa-file-pdf';
                if (file.includes('.pptx') || file.includes('.ppt')) icon = 'fa-file-powerpoint';
                else if (file.includes('.zip')) icon = 'fa-file-archive';
                else if (file.includes('.docx') || file.includes('.doc')) icon = 'fa-file-word';
                else if (file.includes('صور')) icon = 'fa-file-image';
                html += `
                    <div class="file-item">
                        <span class="file-name"><i class="fas ${icon}"></i> ${file}</span>
                        <button class="download-btn" onclick="alert('جاري تحميل: ${file} (محاكاة)')"><i class="fas fa-download"></i> تحميل</button>
                    </div>
                `;
            });
            container.innerHTML = html;
        }

                    loadPortfolioContent(content, `B${i}`);

                container.appendChild(btn);
            }
            loadPortfolioContent(content, 'B1');
        }
